CREATE DATABASE medigestion_db;
USE medigestion_db;

SHOW DATABASES;

CREATE TABLE usuario (
    id_user INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(20) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    full_name VARCHAR(50) NOT NULL,
    rol ENUM ('administrador', 'medico', 'recepcionista', 'farmaceutico') NOT NULL DEFAULT 'medico',
    state ENUM ('activo', 'inactivo')
); 

CREATE TABLE sangre (
    id_tipo INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre_tipo VARCHAR(20) NOT NULL
);

CREATE TABLE paciente (
    id_paciente VARCHAR(10) PRIMARY KEY,
    id_expediente INT UNSIGNED AUTO_INCREMENT UNIQUE,
    full_name VARCHAR(50) NOT NULL,
    fecha_nacimiento DATE NOT NULL,
    sexo ENUM ('masculino', 'femenino') NOT NULL,
    telefono VARCHAR(10) NOT NULL,
    direccion VARCHAR(50),
    id_sangre INT UNSIGNED NOT NULL,
    nombre_seguro_med VARCHAR(50),

    FOREIGN KEY (id_sangre) REFERENCES sangre(id_tipo)
);

CREATE TABLE especialidad(
    id_especialidad INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(20) NOT NULL
);

CREATE TABLE medico(
    id_medico INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    full_name VARCHAR(50) NOT NULL,
    licencia_medica INT UNSIGNED UNIQUE NOT NULL,
    id_especialidad INT NOT NULL,
    telefono VARCHAR(10) NOT NULL,
    email VARCHAR(20),

    FOREIGN KEY (id_especialidad) REFERENCES especialidad(id_especialidad)
);

CREATE TABLE cita_medica(
    id_cita INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    id_paciente VARCHAR(10) NOT NULL,
    id_medico INT UNSIGNED NOT NULL,
    fecha_hora DATETIME NOT NULL,
    motivo VARCHAR(50) NOT NULL,
    estado ENUM('pendiente', 'atendida', 'cancelada'),

    FOREIGN KEY (id_paciente) REFERENCES paciente (id_paciente),
    FOREIGN KEY (id_medico) REFERENCES medico (id_medico)
);

CREATE TABLE categoria_prod(
    id_categoria INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    nombre_prod VARCHAR(20) NOT NULL
);

CREATE TABLE inventario(
    id_producto INT AUTO_INCREMENT PRIMARY KEY,
    nombre_prod VARCHAR(20) NOT NULL,
    id_categoria INT UNSIGNED NOT NULL,
    precio DOUBLE NOT NULL,
    stock INT NOT NULL,
    stock_minimo INT NOT NULL,
    fecha_vencimiento DATE NOT NULL,

    FOREIGN KEY (id_categoria) REFERENCES categoria_prod(id_categoria)
);

CREATE TABLE fac_dato(
    id_factura INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    id_paciente VARCHAR(10) NOT NULL,
    id_user INT UNSIGNED NOT NULL,
    fecha_hora_emision DATETIME NOT NULL,
    tipo_pago ENUM('efectivo', 'tarjeta', 'seguro_medico') NOT NULL,
    porcentaje_seguro INT UNSIGNED NOT NULL,
    subtotal DOUBLE NOT NULL,
    descuento_total INT NOT NULL,
    total_pago DOUBLE NOT NULL,

    FOREIGN KEY (id_paciente) REFERENCES paciente(id_paciente),
    FOREIGN KEY (id_user) REFERENCES usuario(id_user)
);

CREATE TABLE fac_detalle(
    id_detalle INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
    id_factura INT UNSIGNED NOT NULL,
    id_producto INT NOT NULL,
    cantidad INT UNSIGNED NOT NULL DEFAULT 1,
    precio_unitario DOUBLE NOT NULL,

    FOREIGN KEY (id_factura) REFERENCES fac_dato(id_factura),
    FOREIGN KEY (id_producto) REFERENCES inventario(id_producto)
)
